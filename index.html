-- Oracle DDL for PulseChat 1-on-1
CREATE TABLE PULSE_USERS (
    USER_ID VARCHAR2(64) PRIMARY KEY,
    FULL_NAME VARCHAR2(128) NOT NULL,
    EMAIL VARCHAR2(255) UNIQUE NOT NULL,
    PASSWORD_HASH VARCHAR2(128) NOT NULL,
    CREATED_AT TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE PULSE_MESSAGES (
    MSG_ID VARCHAR2(64) PRIMARY KEY,
    SENDER_ID VARCHAR2(64) NOT NULL,
    RECEIVER_ID VARCHAR2(64) NOT NULL,
    MSG_TYPE VARCHAR2(16) NOT NULL, -- 'text' or 'image'
    CONTENT VARCHAR2(4000),
    PHOTO_PAYLOAD CLOB,            -- Base64 photo storage
    CREATED_AT NUMBER NOT NULL
);
CREATE INDEX IDX_CONV ON PULSE_MESSAGES(SENDER_ID, RECEIVER_ID);
